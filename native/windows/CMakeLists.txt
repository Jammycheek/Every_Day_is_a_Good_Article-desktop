cmake_minimum_required(VERSION 3.20)
project(wpd_probe LANGUAGES CXX)
add_executable(wpd-probe main.cpp)
target_compile_features(wpd-probe PRIVATE cxx_std_17)
target_compile_definitions(wpd-probe PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(wpd-probe PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(wpd-probe PRIVATE ole32 PortableDeviceGUIDs cfgmgr32 bcrypt)
enable_testing()
add_test(NAME stream-validation COMMAND wpd-probe self-test)
add_test(NAME invalid-command COMMAND wpd-probe invalid)
set_tests_properties(invalid-command PROPERTIES WILL_FAIL TRUE)
